# LINK TEST: unit a sends to unit b at speed 1 (Sega's rate), then at speed 4; both save what they counted.
run 1.5
expect a 3 NO CABLE
cable on
run 1
expect a 3 CABLE OK
expect b 3 CABLE OK
press a A 0.3
run 25
shot a sent
shot b got
expect a 3 SENT OK!
expect a 1 SENT 001
expect b 3 GOT IT OK!
expect b 2 GOT  001
# speed 4 on both
press a U 0.6
run 0.6
press a U 0.6
run 0.6
press a U 0.6
run 0.6
press b U 0.6
run 0.6
press b U 0.6
run 0.6
press b U 0.6
run 1
expect a 4 SPEED 4 XT
expect b 4 SPEED 4 XT
press a A 0.3
run 6
expect a 3 SENT OK!
expect a 1 SENT 002
expect b 2 GOT  002
# b sends back
press b A 0.3
run 6
expect b 3 SENT OK!
expect a 2 GOT  001
# speed 8: the link on the RC clock
press a D 0.6
run 0.6
press b D 0.6
run 0.6
press a D 0.6
run 0.6
press b D 0.6
run 0.6
press a D 0.6
run 0.6
press b D 0.6
run 0.6
press a D 0.6
run 0.6
press b D 0.6
run 1
expect a 4 SPEED 8 RC
expect b 4 SPEED 8 RC
press a A 0.3
run 4
expect a 3 SENT OK!
expect b 2 GOT  003
# one raw byte (B): the other side shows it
press a B 0.3
run 2
expect b 3 RAW A0 001
press a B 0.3
run 2
expect b 3 RAW A1 002
save a OUT/a.bin
save b OUT/b.bin
cable off
run 1.5
